class Krug{
    public:
        Tacka centar;
        int precnik;
    Krug( double r = 1.0 , double cx = 0.0 , double cy = 0.0){
        precnik = r;
        centar.x = cx;
        centar.y = cy; 

    
    }
    double obim()const{
        return 2*precnik * 3.14;
    }
    double povrsina()const{
        return 3.14 * (precnik * precnik);
    }

};
