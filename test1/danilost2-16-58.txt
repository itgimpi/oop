#include <bits/stdc++.h>
using namespace std;

class Auto {
public:
    string marka;
    int godina;
    int kilometraza;
    int snaga;

    Auto(string m = "", int god = 0, int km = 0, int p = 0) {
        marka = m;
        godina = god;
        kilometraza = km;
        snaga = p;
    }

    void ispis() {
        cout << marka << endl;
        cout << "Godiste: " << godina << endl;
        cout << "Kilometraza: " << kilometraza << " km" << endl;
        cout << "Snaga: " << snaga << " KS" << endl;
    }
};

int main() {
    Auto a1("BMW", 2015, 150000, 250);
    a1.ispis();

    return 0;
}
